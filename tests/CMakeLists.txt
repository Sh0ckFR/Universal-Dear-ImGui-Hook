function(uih_add_host_test name)
    add_test(NAME ${name} COMMAND uih_host ${ARGN})
    set_tests_properties(${name} PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 120 RUN_SERIAL TRUE)
endfunction()

if(WIN32)
    add_executable(uih_host host.cpp host_vulkan.cpp)
    target_link_libraries(uih_host PRIVATE uih d3d9 d3d10 d3d11 d3d12 dxgi opengl32 dinput8 dxguid)
    target_compile_definitions(uih_host PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)

    # A controller the tests drive, in its own directory so that only the
    # gamepad scenario loads it in place of the system xinput1_4.dll.
    add_library(uih_fake_xinput SHARED fake_xinput.cpp fake_xinput.def)
    set_target_properties(uih_fake_xinput PROPERTIES OUTPUT_NAME xinput1_4 PREFIX ""
                          RUNTIME_OUTPUT_DIRECTORY $<1:${CMAKE_CURRENT_BINARY_DIR}/fake_xinput>)
    if(MINGW)
        target_link_options(uih_fake_xinput PRIVATE -Wl,--enable-stdcall-fixup)
    endif()

    uih_add_host_test(d3d9 --api d3d9)
    uih_add_host_test(d3d9ex --api d3d9ex)
    uih_add_host_test(d3d10 --api d3d10)
    uih_add_host_test(d3d11 --api d3d11)
    uih_add_host_test(d3d11_early --api d3d11 --inject-at 0)
    uih_add_host_test(d3d11_scale --api d3d11 --scale 2)
    uih_add_host_test(d3d12 --api d3d12)
    uih_add_host_test(d3d12_decoy --api d3d12 --decoy-queue)
    uih_add_host_test(d3d12_decoy_early --api d3d12 --decoy-queue --inject-at 0)
    uih_add_host_test(d3d12_remove_device --api d3d12 --remove-device --capture)
    uih_add_host_test(vulkan --api vulkan)
    uih_add_host_test(vulkan_early --api vulkan --inject-at 0)
    uih_add_host_test(vulkan_transfer_only --api vulkan --transfer-only --capture)
    # The swap chain was created before the hook, and the host ignores VK_SUBOPTIMAL_KHR.
    uih_add_host_test(vulkan_ignore_suboptimal --api vulkan --ignore-suboptimal --capture)
    uih_add_host_test(vulkan_second_window_shared --api vulkan --second-window --shared-device)
    # Frames of the overlay are still in flight when it moves to the other swap chain.
    uih_add_host_test(d3d12_second_window_shared --api d3d12 --second-window --shared-device --gpu-latency 50)
    uih_add_host_test(opengl --api opengl)
    foreach(api d3d9 d3d9ex d3d10 d3d11 d3d12 vulkan opengl)
        uih_add_host_test(${api}_resize --api ${api} --resize --capture)
        uih_add_host_test(${api}_reload --api ${api} --reload --capture)
        uih_add_host_test(${api}_second_window --api ${api} --second-window)
        # More textures than CreateTexture() accepts, all drawn.
        uih_add_host_test(${api}_textures --api ${api} --textures 300 --capture)
        # Skipped unless UIH_TEST_FULLSCREEN is set: it changes the display mode.
        uih_add_host_test(${api}_fullscreen --api ${api} --fullscreen)
    endforeach()
    uih_add_host_test(remove --api d3d11 --remove)
    # Recovery from a crash of the overlay needs structured exception handling.
    if(MSVC)
        foreach(api d3d9 d3d10 d3d11 d3d12 vulkan opengl)
            uih_add_host_test(${api}_fault --api ${api} --fault access)
        endforeach()
        # Not D3D9: the state block of its backend leaks, and the device cannot be reset.
        foreach(api d3d10 d3d11 d3d12 vulkan opengl)
            uih_add_host_test(${api}_fault_draw --api ${api} --fault draw)
        endforeach()
        uih_add_host_test(d3d11_fault_exception --api d3d11 --fault exception)
    endif()
    uih_add_host_test(d3d11_resubclass --api d3d11 --reload --resubclass)
    uih_add_host_test(opengl_recreate_context --api opengl --recreate-context --capture)
    uih_add_host_test(opengl_then_d3d11 --api opengl --switch-api d3d11)
    uih_add_host_test(opengl_bound_framebuffer --api opengl --bound-framebuffer --capture)
    foreach(api d3d9 d3d11 d3d12 vulkan opengl)
        uih_add_host_test(${api}_slow_frame --api ${api} --slow-frame 2500)
        uih_add_host_test(${api}_stuck_frame --api ${api} --slow-frame 6000)
    endforeach()
    foreach(api d3d11 opengl)
        uih_add_host_test(${api}_render_thread --api ${api} --render-thread)
    endforeach()
    uih_add_host_test(input_raw --api d3d11 --input raw)
    uih_add_host_test(input_directinput --api d3d11 --input directinput)
    uih_add_host_test(gamepad --api d3d11 --gamepad $<TARGET_FILE:uih_fake_xinput>)

    # Back buffers storing linear values: the overlay colors must come out unchanged.
    uih_add_host_test(d3d9_srgb --api d3d9 --format srgb --capture)
    uih_add_host_test(d3d10_srgb --api d3d10 --format srgb --capture)
    uih_add_host_test(d3d11_srgb --api d3d11 --format srgb --capture)
    uih_add_host_test(d3d11_float --api d3d11 --format float --capture)
    uih_add_host_test(d3d12_float --api d3d12 --format float --capture)
    uih_add_host_test(vulkan_srgb --api vulkan --format srgb --capture)
    uih_add_host_test(vulkan_srgb_early --api vulkan --format srgb --capture --inject-at 0 --settle 3000)
    uih_add_host_test(opengl_srgb --api opengl --format srgb --capture)

    if(TARGET uih_example)
        uih_add_host_test(dll_d3d11 --api d3d11 --dll $<TARGET_FILE:uih_example>)
        uih_add_host_test(dll_vulkan --api vulkan --dll $<TARGET_FILE:uih_example>)
        uih_add_host_test(dll_opengl --api opengl --dll $<TARGET_FILE:uih_example>)
        uih_add_host_test(dll_d3d9 --api d3d9 --dll $<TARGET_FILE:uih_example>)
        uih_add_host_test(dll_proxy --proxy $<TARGET_FILE:uih_example>)
        uih_add_host_test(dll_reload --api d3d11 --dll $<TARGET_FILE:uih_example> --reload)
        uih_add_host_test(dll_resubclass --api d3d11 --dll $<TARGET_FILE:uih_example> --resubclass)

        # The DLL under the name a game loads it as a proxy.
        set(UIH_PROXY_DLL ${CMAKE_CURRENT_BINARY_DIR}/proxy/dinput8.dll)
        add_custom_target(uih_proxy_dll ALL
            COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:uih_example> ${UIH_PROXY_DLL}
            DEPENDS uih_example)
        uih_add_host_test(dll_proxy_reload --api d3d11 --dll ${UIH_PROXY_DLL} --reload)
    endif()
else()
    pkg_check_modules(UIH_HOST REQUIRED IMPORTED_TARGET x11 xtst gl egl)

    # The host does not link the hook: the tests preload the example library.
    add_executable(uih_host host_linux.cpp host_vulkan.cpp)
    target_include_directories(uih_host PRIVATE ${UIH_THIRD_PARTY}/vulkan/include)
    target_compile_definitions(uih_host PRIVATE VK_NO_PROTOTYPES)
    target_link_libraries(uih_host PRIVATE PkgConfig::UIH_HOST ${CMAKE_DL_LIBS})

    if(TARGET uih_example)
        set(UIH_PRELOAD $<TARGET_FILE:uih_example>)
        set(UIH_SANITIZER_OPTIONS)
        if(UIH_SANITIZE)
            # The sanitizer runtime must be loaded before the library it checks.
            execute_process(COMMAND ${CMAKE_CXX_COMPILER} -print-file-name=libasan.so
                            OUTPUT_VARIABLE UIH_ASAN_RUNTIME OUTPUT_STRIP_TRAILING_WHITESPACE)
            set(UIH_PRELOAD ${UIH_ASAN_RUNTIME}:${UIH_PRELOAD})
            # No leak check: the host leaves its X11 and GPU objects to the end of the process.
            set(UIH_SANITIZER_OPTIONS
                "ASAN_OPTIONS=detect_leaks=0"
                "UBSAN_OPTIONS=print_stacktrace=1")
        endif()

        function(uih_add_preload_test name)
            uih_add_host_test(${name} ${ARGN})
            set_tests_properties(${name} PROPERTIES ENVIRONMENT "LD_PRELOAD=${UIH_PRELOAD};UIH_DEBUG=1;${UIH_SANITIZER_OPTIONS}")
        endfunction()

        uih_add_preload_test(preload_glx --api opengl)
        uih_add_preload_test(preload_glx_lookup --api opengl --lookup)
        uih_add_preload_test(preload_egl --api opengl --egl)
        uih_add_preload_test(preload_egl_lookup --api opengl --egl --lookup)
        uih_add_preload_test(preload_vulkan --api vulkan)
        uih_add_preload_test(preload_vulkan_device --api vulkan --device-proc-addr)
        uih_add_preload_test(preload_vulkan_transfer_only --api vulkan --transfer-only)
        uih_add_preload_test(preload_glx_resize --api opengl --resize)
        uih_add_preload_test(preload_egl_resize --api opengl --egl --resize)
        uih_add_preload_test(preload_glx_recreate_context --api opengl --recreate-context)
        uih_add_preload_test(preload_egl_recreate_context --api opengl --egl --recreate-context)
        uih_add_preload_test(preload_glx_bound_framebuffer --api opengl --bound-framebuffer)
        uih_add_preload_test(preload_egl_bound_framebuffer --api opengl --egl --bound-framebuffer)
        uih_add_preload_test(preload_egl_swap_with_damage --api opengl --egl --swap-with-damage)
        uih_add_preload_test(preload_vulkan_resize --api vulkan --resize)
        uih_add_preload_test(preload_glx_reload --api opengl --reload)
        uih_add_preload_test(preload_vulkan_reload --api vulkan --reload)
        uih_add_preload_test(preload_glx_second_window --api opengl --second-window)
        uih_add_preload_test(preload_vulkan_second_window --api vulkan --second-window)
        # The Vulkan loader opened as libvulkan.so, the name the development package installs.
        find_library(UIH_VULKAN_LOADER NAMES libvulkan.so.1)
        if(UIH_VULKAN_LOADER)
            set(unversioned ${CMAKE_CURRENT_BINARY_DIR}/unversioned/libvulkan.so)
            file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/unversioned)
            file(CREATE_LINK ${UIH_VULKAN_LOADER} ${unversioned} SYMBOLIC)
            uih_add_preload_test(preload_vulkan_unversioned --api vulkan --vulkan-library ${unversioned})
        endif()
    endif()
endif()
